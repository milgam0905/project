cmake_minimum_required(VERSION 3.16)
project(blood_principle_module LANGUAGES CXX)
option(BP_WITH_SQLITE "Build read-only content loader" ON)
option(BP_BUILD_TESTS "Build module tests" OFF)
option(BP_BUILD_EXAMPLE "Build console example" OFF)
add_library(bp_sim STATIC src/simulation.cpp)
add_library(bp::sim ALIAS bp_sim)
target_include_directories(bp_sim PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(bp_sim PUBLIC cxx_std_17)
if(BP_WITH_SQLITE)
  set(BP_SQLITE_SOURCE_DIR "" CACHE PATH "Optional directory containing sqlite3.c and sqlite3.h")
  if(BP_SQLITE_SOURCE_DIR)
    enable_language(C)
    add_library(bp_sqlite STATIC "${BP_SQLITE_SOURCE_DIR}/sqlite3.c")
    target_include_directories(bp_sqlite PUBLIC "${BP_SQLITE_SOURCE_DIR}")
  else()
    find_package(SQLite3 REQUIRED)
  endif()
  add_library(bp_content STATIC src/content.cpp)
  add_library(bp::content ALIAS bp_content)
  target_include_directories(bp_content PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(bp_content PUBLIC cxx_std_17)
  if(TARGET bp_sqlite)
    target_link_libraries(bp_content PUBLIC bp_sim bp_sqlite)
  elseif(TARGET SQLite3::SQLite3)
    target_link_libraries(bp_content PUBLIC bp_sim SQLite3::SQLite3)
  else()
    target_link_libraries(bp_content PUBLIC bp_sim SQLite::SQLite3)
  endif()
  add_library(bp_native STATIC src/native.cpp)
  add_library(bp::native ALIAS bp_native)
  target_include_directories(bp_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(bp_native PUBLIC cxx_std_17)
  if(TARGET bp_sqlite)
    target_link_libraries(bp_native PUBLIC bp_sqlite)
  elseif(TARGET SQLite3::SQLite3)
    target_link_libraries(bp_native PUBLIC SQLite3::SQLite3)
  else()
    target_link_libraries(bp_native PUBLIC SQLite::SQLite3)
  endif()
endif()
if(BP_BUILD_TESTS)
  enable_testing()
  add_executable(bp_tests tests/simulation_tests.cpp)
  target_link_libraries(bp_tests PRIVATE bp_sim)
  add_test(NAME bp_simulation COMMAND bp_tests)
  if(BP_WITH_SQLITE)
    add_executable(bp_content_tests tests/content_tests.cpp)
    target_link_libraries(bp_content_tests PRIVATE bp_content)
    add_test(NAME bp_content COMMAND bp_content_tests)
    add_executable(bp_native_tests tests/native_tests.cpp)
    target_link_libraries(bp_native_tests PRIVATE bp_native)
    add_test(NAME bp_native COMMAND bp_native_tests "${CMAKE_CURRENT_SOURCE_DIR}/data/blood_principle_game.db")
  endif()
endif()
if(BP_BUILD_EXAMPLE)
  add_executable(bp_example examples/main.cpp)
  target_link_libraries(bp_example PRIVATE bp_sim)
  if(BP_WITH_SQLITE)
    add_executable(bp_native_example examples/native_main.cpp)
    target_link_libraries(bp_native_example PRIVATE bp_native)
  endif()
endif()
